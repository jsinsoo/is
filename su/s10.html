"""
오픈소스·라이브러리 활용 협력 프로젝트 예시
- Pandas를 이용한 데이터 전처리와 검색
- Matplotlib을 이용한 그래프 2종 생성
- 한글 글꼴과 음수 기호 설정
- 오류 처리, 출처·라이선스 표시
- CSV와 PNG 산출물 저장
"""

from pathlib import Path
import platform
import pandas as pd
import matplotlib.pyplot as plt

# --------------------------------------------------
# 1. Matplotlib 한글 글꼴 설정
# --------------------------------------------------
if platform.system() == "Windows":
    plt.rcParams["font.family"] = "Malgun Gothic"
elif platform.system() == "Darwin":
    plt.rcParams["font.family"] = "AppleGothic"
else:
    # 리눅스·코랩에서 나눔고딕이 설치된 경우 사용한다.
    plt.rcParams["font.family"] = "NanumGothic"

# 음수 기호가 네모로 깨지는 현상을 방지한다.
plt.rcParams["axes.unicode_minus"] = False


# --------------------------------------------------
# 2. 출력 폴더와 자료 정보
# --------------------------------------------------
OUTPUT_DIR = Path("school_dashboard_output")
OUTPUT_DIR.mkdir(exist_ok=True)

DATA_SOURCE = "수업용 가상 학교생활 데이터"
DATA_LICENSE = "교육 목적 자유 이용, 재배포 시 출처 표시"
LIBRARIES = {
    "pandas": "BSD 3-Clause License",
    "matplotlib": "PSF-based License",
}


# --------------------------------------------------
# 3. 가상 데이터 생성
# --------------------------------------------------
sample_data = pd.DataFrame({
    "date": [
        "2026-07-01",
        "2026-07-02",
        "2026-07-03",
        "2026-07-04",
        "2026-07-05",
        "2026-07-06",
    ],
    "category": [
        "급식",
        "공지",
        "행사",
        "공지",
        "급식",
        "행사",
    ],
    "title": [
        "7월 식단 안내",
        "과학실 안전교육",
        "수학·과학 체험전",
        "도서관 운영 시간",
        None,
        "동아리 발표회",
    ],
    "views": [120, 85, 210, None, 98, 175],
    "importance": [3, 5, 4, 2, 3, 5],
})

sample_csv = OUTPUT_DIR / "school_data.csv"

sample_data.to_csv(
    sample_csv,
    index=False,
    encoding="utf-8-sig",
)


# --------------------------------------------------
# 4. 데이터 불러오기
# --------------------------------------------------
def load_data(path: str | Path) -> pd.DataFrame:
    """CSV 파일을 안전하게 불러온다."""
    path = Path(path)

    if not path.exists():
        raise FileNotFoundError(
            f"파일을 찾을 수 없습니다: {path}"
        )

    try:
        return pd.read_csv(path)
    except pd.errors.EmptyDataError as error:
        raise ValueError(
            "데이터 파일이 비어 있습니다."
        ) from error
    except UnicodeDecodeError:
        return pd.read_csv(
            path,
            encoding="cp949",
        )


# --------------------------------------------------
# 5. 데이터 전처리
# --------------------------------------------------
def preprocess(df: pd.DataFrame) -> pd.DataFrame:
    """필수 열을 확인하고 결측값과 자료형을 정리한다."""
    required_columns = {
        "date",
        "category",
        "title",
        "views",
        "importance",
    }

    missing_columns = (
        required_columns - set(df.columns)
    )

    if missing_columns:
        raise ValueError(
            "필수 열이 없습니다: "
            f"{sorted(missing_columns)}"
        )

    cleaned = df.copy()

    cleaned["date"] = pd.to_datetime(
        cleaned["date"],
        errors="coerce",
    )

    cleaned["title"] = (
        cleaned["title"]
        .fillna("제목 없음")
    )

    cleaned["views"] = pd.to_numeric(
        cleaned["views"],
        errors="coerce",
    )

    cleaned["importance"] = pd.to_numeric(
        cleaned["importance"],
        errors="coerce",
    )

    cleaned["views"] = (
        cleaned["views"]
        .fillna(cleaned["views"].median())
    )

    cleaned["importance"] = (
        cleaned["importance"]
        .fillna(3)
        .clip(1, 5)
    )

    cleaned = cleaned.dropna(
        subset=["date", "category"]
    )

    return cleaned


# --------------------------------------------------
# 6. 조건 검색
# --------------------------------------------------
def search_data(
    df: pd.DataFrame,
    keyword: str = "",
    category: str = "",
    min_importance: int = 1,
) -> pd.DataFrame:
    """검색 조건을 만족하는 자료를 반환한다."""
    result = df.copy()

    if keyword:
        result = result[
            result["title"].str.contains(
                keyword,
                case=False,
                na=False,
            )
        ]

    if category:
        result = result[
            result["category"] == category
        ]

    result = result[
        result["importance"] >= min_importance
    ]

    return result.sort_values(
        ["importance", "views"],
        ascending=[False, False],
    )


# --------------------------------------------------
# 7. 시각화
# --------------------------------------------------
def visualize(
    df: pd.DataFrame,
) -> list[Path]:
    """분류별 자료 수와 날짜별 조회 수 그래프를 저장한다."""
    graph_paths: list[Path] = []

    # 그래프 1: 분류별 자료 수
    category_counts = (
        df["category"]
        .value_counts()
    )

    plt.figure(figsize=(8, 5))
    category_counts.plot(kind="bar")
    plt.xlabel("분류")
    plt.ylabel("자료 수")
    plt.title("분류별 학교생활 자료 수")
    plt.xticks(rotation=0)
    plt.tight_layout()

    graph1 = (
        OUTPUT_DIR
        / "category_counts.png"
    )

    plt.savefig(
        graph1,
        dpi=150,
    )
    plt.close()

    graph_paths.append(graph1)

    # 그래프 2: 날짜별 조회 수
    daily_views = (
        df.groupby(
            "date",
            as_index=False,
        )["views"]
        .sum()
    )

    plt.figure(figsize=(9, 5))
    plt.plot(
        daily_views["date"],
        daily_views["views"],
        marker="o",
    )
    plt.xlabel("날짜")
    plt.ylabel("조회 수")
    plt.title("날짜별 학교생활 자료 조회 수")
    plt.xticks(rotation=45)
    plt.tight_layout()

    graph2 = (
        OUTPUT_DIR
        / "daily_views.png"
    )

    plt.savefig(
        graph2,
        dpi=150,
    )
    plt.close()

    graph_paths.append(graph2)

    return graph_paths


# --------------------------------------------------
# 8. 프로젝트 실행
# --------------------------------------------------
def main() -> None:
    try:
        raw_data = load_data(sample_csv)
        cleaned_data = preprocess(raw_data)

        # 중요도 3 이상 자료 검색
        search_result = search_data(
            cleaned_data,
            min_importance=3,
        )

        cleaned_path = (
            OUTPUT_DIR
            / "cleaned_school_data.csv"
        )

        result_path = (
            OUTPUT_DIR
            / "search_result.csv"
        )

        cleaned_data.to_csv(
            cleaned_path,
            index=False,
            encoding="utf-8-sig",
        )

        search_result.to_csv(
            result_path,
            index=False,
            encoding="utf-8-sig",
        )

        graph_paths = visualize(
            cleaned_data
        )

        print("[검색 결과]")
        print(
            search_result.to_string(
                index=False
            )
        )

        print("\n[생성된 산출물]")
        print("원본 데이터:", sample_csv)
        print("정제 데이터:", cleaned_path)
        print("검색 결과:", result_path)

        for graph_path in graph_paths:
            print("그래프:", graph_path)

        print("\n[자료 출처·라이선스]")
        print("자료 출처:", DATA_SOURCE)
        print("자료 이용 조건:", DATA_LICENSE)

        print("\n[사용 라이브러리]")
        for name, license_name in LIBRARIES.items():
            print(
                f"{name}: {license_name}"
            )

    except (
        FileNotFoundError,
        ValueError,
    ) as error:
        print(
            "처리 중 오류:",
            error,
        )


if __name__ == "__main__":
    main()

12. 예시 정답

문항 1 정답
Pandas는 CSV 불러오기, 자료형 변환, 결측값 처리, 조건 검색, 집계에 사용한다. Matplotlib은 분석 결과를 막대그래프와 선그래프로 표현하고 PNG 파일로 저장하는 데 사용한다.
문항 2 정답
font.family 설정은 그래프의 제목, 축 이름, 범례에서 한글이 깨지는 것을 방지한다. axes.unicode_minus를 False로 설정하면 일부 한글 글꼴에서 음수 기호가 네모로 표시되는 문제를 줄일 수 있다.
문항 3 정답
오픈소스는 자유롭게 사용할 수 있더라도 라이선스마다 출처 표시, 수정본 배포, 저작권 고지 등의 조건이 다르다. 조건을 확인하지 않으면 저작권과 배포 규정을 위반할 수 있다.
문항 4 정답
변경 기록은 오류 발생 원인을 추적하고 이전 상태로 되돌리는 데 도움이 된다. 개인별 기여 기록은 역할 수행과 협업 과정을 공정하게 평가하고, 특정 팀원에게 업무가 집중되었는지 확인하는 근거가 된다.
문항 5 정답
개인정보가 포함된 학교 자료를 그대로 사용하거나 조회 수를 학생 선호도로 단정하는 문제가 생길 수 있다. 개인정보를 제거하고 필요한 열만 사용하며, 데이터 출처와 수집 목적을 공개하고, 분석 결과의 한계와 편향 가능성을 함께 제시해야 한다.
