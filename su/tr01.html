재귀 선분 나선

기초

추천 실행 환경 Python IDLE · VS Code · 로컬 Python 
import turtle

t = turtle.Turtle()
t.speed(0)

def spiral(length):
    if length < 5:
        return
    t.forward(length)
    t.left(90)
    spiral(length - 5)

spiral(200)
turtle.done()

코흐 곡선

응용

하나의 선분을 네 개의 작은 선분으로 바꾸는 프랙탈입니다.

추천 실행 환경 Python IDLE · VS Code · 로컬 Python 
import turtle

t = turtle.Turtle()
t.speed(0)

def koch(length, depth):
    if depth == 0:
        t.forward(length)
        return

    length /= 3

    koch(length, depth - 1)
    t.left(60)

    koch(length, depth - 1)
    t.right(120)

    koch(length, depth - 1)
    t.left(60)

    koch(length, depth - 1)

koch(500, 4)
turtle.done()
코드 학습 포인트

t.speed(0)

def plant(length, depth):
    if depth == 0:
        t.dot(5)
        return

    t.forward(length)

    t.left(25)
    plant(length * 0.72, depth - 1)

    t.right(50)
    plant(length * 0.72, depth - 1)

    t.left(25)
    t.backward(length)

for x in [-220, 0, 220]:
    t.penup()
    t.goto(x, -260)
    t.setheading(90)
    t.pendown()

    plant(85, 7)

turtle.done()
