from dataclasses import dataclass
from typing import Any, Iterable


# 1. 한 개의 예측 결과
@dataclass
class PredictionResult:
    actual: Any
    predicted: Any
    positive_probability: float | None = None


# 2. 양성·음성 클래스 정의
@dataclass
class BinaryLabel:
    positive_label: Any
    negative_label: Any

    def to_binary(self, label: Any) -> str:
        return "양성" if label == self.positive_label else "음성"


# 3. 이진 혼동행렬
class BinaryConfusionMatrix:
    def __init__(self, labels: BinaryLabel):
        self.labels = labels
        self.tp = 0
        self.tn = 0
        self.fp = 0
        self.fn = 0

    def reset(self) -> None:
        self.tp = self.tn = self.fp = self.fn = 0

    def update(self, actual: Any, predicted: Any) -> None:
        actual_binary = self.labels.to_binary(actual)
        predicted_binary = self.labels.to_binary(predicted)

        if actual_binary == "양성" and predicted_binary == "양성":
            self.tp += 1
        elif actual_binary == "음성" and predicted_binary == "음성":
            self.tn += 1
        elif actual_binary == "음성" and predicted_binary == "양성":
            self.fp += 1
        else:
            self.fn += 1

    def build(self, results: Iterable[PredictionResult]) -> None:
        self.reset()
        for result in results:
            self.update(result.actual, result.predicted)

    @property
    def total(self) -> int:
        return self.tp + self.tn + self.fp + self.fn

    def as_dict(self) -> dict[str, int]:
        return {
            "TP": self.tp,
            "TN": self.tn,
            "FP": self.fp,
            "FN": self.fn,
        }


# 4. 안전한 나눗셈
class SafeDivision:
    @staticmethod
    def divide(numerator: float, denominator: float) -> float:
        return numerator / denominator if denominator else 0.0


# 5. 이진 분류 평가 지표
class BinaryMetricCalculator:
    def __init__(self, matrix: BinaryConfusionMatrix):
        self.matrix = matrix

    def accuracy(self) -> float:
        return SafeDivision.divide(
            self.matrix.tp + self.matrix.tn,
            self.matrix.total
        )

    def precision(self) -> float:
        return SafeDivision.divide(
            self.matrix.tp,
            self.matrix.tp + self.matrix.fp
        )

    def recall(self) -> float:
        return SafeDivision.divide(
            self.matrix.tp,
            self.matrix.tp + self.matrix.fn
        )

    def specificity(self) -> float:
        return SafeDivision.divide(
            self.matrix.tn,
            self.matrix.tn + self.matrix.fp
        )

    def false_positive_rate(self) -> float:
        return SafeDivision.divide(
            self.matrix.fp,
            self.matrix.fp + self.matrix.tn
        )

    def false_negative_rate(self) -> float:
        return SafeDivision.divide(
            self.matrix.fn,
            self.matrix.fn + self.matrix.tp
        )

    def negative_predictive_value(self) -> float:
        return SafeDivision.divide(
            self.matrix.tn,
            self.matrix.tn + self.matrix.fn
        )

    def f1_score(self) -> float:
        precision = self.precision()
        recall = self.recall()
        return SafeDivision.divide(
            2 * precision * recall,
            precision + recall
        )

    def balanced_accuracy(self) -> float:
        return (self.recall() + self.specificity()) / 2

    def all_metrics(self) -> dict[str, float]:
        return {
            "accuracy": self.accuracy(),
            "precision": self.precision(),
            "recall": self.recall(),
            "specificity": self.specificity(),
            "false_positive_rate": self.false_positive_rate(),
            "false_negative_rate": self.false_negative_rate(),
            "negative_predictive_value":
                self.negative_predictive_value(),
            "f1_score": self.f1_score(),
            "balanced_accuracy": self.balanced_accuracy(),
        }


# 6. 임계값 기반 분류기
class ThresholdClassifier:
    def __init__(self,
                 positive_label: Any,
                 negative_label: Any,
                 threshold: float = 0.5):
        self.positive_label = positive_label
        self.negative_label = negative_label
        self.threshold = threshold

    def predict(self, probability: float) -> Any:
        if probability >= self.threshold:
            return self.positive_label
        return self.negative_label

    def predict_many(self,
                     probabilities: Iterable[float]) -> list[Any]:
        return [
            self.predict(probability)
            for probability in probabilities
        ]


# 7. 여러 임계값의 성능 비교
class ThresholdAnalyzer:
    def __init__(self,
                 labels: BinaryLabel,
                 probabilities: list[float],
                 actual: list[Any]):
        self.labels = labels
        self.probabilities = probabilities
        self.actual = actual

    def evaluate(self, threshold: float) -> dict[str, float]:
        classifier = ThresholdClassifier(
            self.labels.positive_label,
            self.labels.negative_label,
            threshold
        )
        predicted = classifier.predict_many(self.probabilities)

        results = [
            PredictionResult(a, p, probability)
            for a, p, probability in zip(
                self.actual,
                predicted,
                self.probabilities
            )
        ]

        matrix = BinaryConfusionMatrix(self.labels)
        matrix.build(results)
        metrics = BinaryMetricCalculator(matrix).all_metrics()

        return {
            "threshold": threshold,
            **matrix.as_dict(),
            **metrics,
        }

    def compare(self,
                thresholds: Iterable[float]) -> list[dict[str, float]]:
        return [
            self.evaluate(threshold)
            for threshold in thresholds
        ]


# 8. 다중 클래스 혼동행렬
class MulticlassConfusionMatrix:
    def __init__(self, labels: list[Any]):
        self.labels = labels
        self.matrix = {
            actual: {
                predicted: 0
                for predicted in labels
            }
            for actual in labels
        }

    def update(self, actual: Any, predicted: Any) -> None:
        self.matrix[actual][predicted] += 1

    def build(self,
              actual_values: Iterable[Any],
              predicted_values: Iterable[Any]) -> None:
        for actual, predicted in zip(
            actual_values,
            predicted_values
        ):
            self.update(actual, predicted)

    def diagonal_sum(self) -> int:
        return sum(
            self.matrix[label][label]
            for label in self.labels
        )

    def total(self) -> int:
        return sum(
            self.matrix[actual][predicted]
            for actual in self.labels
            for predicted in self.labels
        )

    def accuracy(self) -> float:
        return SafeDivision.divide(
            self.diagonal_sum(),
            self.total()
        )


# 9. One-vs-Rest 변환과 클래스별 평가
class OneVsRestEvaluator:
    def __init__(self,
                 actual: list[Any],
                 predicted: list[Any],
                 labels: list[Any]):
        self.actual = actual
        self.predicted = predicted
        self.labels = labels

    def evaluate_class(self, target: Any) -> dict[str, float]:
        binary_labels = BinaryLabel(
            positive_label=target,
            negative_label="나머지"
        )

        converted_results = [
            PredictionResult(
                actual=(
                    target if actual == target else "나머지"
                ),
                predicted=(
                    target if predicted == target else "나머지"
                )
            )
            for actual, predicted in zip(
                self.actual,
                self.predicted
            )
        ]

        matrix = BinaryConfusionMatrix(binary_labels)
        matrix.build(converted_results)

        metrics = BinaryMetricCalculator(matrix).all_metrics()
        support = sum(
            actual == target
            for actual in self.actual
        )

        return {
            "label": target,
            "support": support,
            **matrix.as_dict(),
            **metrics,
        }

    def evaluate_all(self) -> list[dict[str, float]]:
        return [
            self.evaluate_class(label)
            for label in self.labels
        ]


# 10. 매크로·가중·마이크로 평균
class AveragingMetric:
    @staticmethod
    def macro_average(
        class_reports: list[dict[str, float]],
        metric_name: str
    ) -> float:
        scores = [
            report[metric_name]
            for report in class_reports
        ]
        return SafeDivision.divide(
            sum(scores),
            len(scores)
        )

    @staticmethod
    def weighted_average(
        class_reports: list[dict[str, float]],
        metric_name: str
    ) -> float:
        total_support = sum(
            report["support"]
            for report in class_reports
        )
        weighted_sum = sum(
            report[metric_name] * report["support"]
            for report in class_reports
        )
        return SafeDivision.divide(
            weighted_sum,
            total_support
        )

    @staticmethod
    def micro_precision(
        class_reports: list[dict[str, float]]
    ) -> float:
        total_tp = sum(report["TP"] for report in class_reports)
        total_fp = sum(report["FP"] for report in class_reports)
        return SafeDivision.divide(
            total_tp,
            total_tp + total_fp
        )

    @staticmethod
    def micro_recall(
        class_reports: list[dict[str, float]]
    ) -> float:
        total_tp = sum(report["TP"] for report in class_reports)
        total_fn = sum(report["FN"] for report in class_reports)
        return SafeDivision.divide(
            total_tp,
            total_tp + total_fn
        )


# 11. 클래스 불균형 분석
class ImbalanceAnalyzer:
    def class_distribution(
        self,
        labels: Iterable[Any]
    ) -> dict[Any, int]:
        distribution: dict[Any, int] = {}
        for label in labels:
            distribution[label] = (
                distribution.get(label, 0) + 1
            )
        return distribution

    def majority_baseline(
        self,
        labels: list[Any]
    ) -> float:
        distribution = self.class_distribution(labels)
        return SafeDivision.divide(
            max(distribution.values()),
            len(labels)
        )

    def imbalance_ratio(
        self,
        labels: list[Any]
    ) -> float:
        counts = list(
            self.class_distribution(labels).values()
        )
        return SafeDivision.divide(
            max(counts),
            min(counts)
        )


# 12. 오류 비용
@dataclass
class ErrorCostPolicy:
    false_positive_cost: float = 1.0
    false_negative_cost: float = 1.0


class CostSensitiveEvaluator:
    def __init__(self,
                 matrix: BinaryConfusionMatrix,
                 policy: ErrorCostPolicy):
        self.matrix = matrix
        self.policy = policy

    def total_cost(self) -> float:
        return (
            self.matrix.fp *
            self.policy.false_positive_cost
            +
            self.matrix.fn *
            self.policy.false_negative_cost
        )

    def average_cost(self) -> float:
        return SafeDivision.divide(
            self.total_cost(),
            self.matrix.total
        )


# 13. 통합 평가 보고서
class ClassificationReport:
    def __init__(self,
                 actual: list[Any],
                 predicted: list[Any],
                 labels: list[Any]):
        self.actual = actual
        self.predicted = predicted
        self.labels = labels

    def create(self) -> dict[str, Any]:
        multiclass_matrix = MulticlassConfusionMatrix(
            self.labels
        )
        multiclass_matrix.build(
            self.actual,
            self.predicted
        )

        evaluator = OneVsRestEvaluator(
            self.actual,
            self.predicted,
            self.labels
        )
        class_reports = evaluator.evaluate_all()

        return {
            "confusion_matrix":
                multiclass_matrix.matrix,
            "accuracy":
                multiclass_matrix.accuracy(),
            "class_reports":
                class_reports,
            "macro_precision":
                AveragingMetric.macro_average(
                    class_reports,
                    "precision"
                ),
            "macro_recall":
                AveragingMetric.macro_average(
                    class_reports,
                    "recall"
                ),
            "macro_f1":
                AveragingMetric.macro_average(
                    class_reports,
                    "f1_score"
                ),
            "weighted_f1":
                AveragingMetric.weighted_average(
                    class_reports,
                    "f1_score"
                ),
            "micro_precision":
                AveragingMetric.micro_precision(
                    class_reports
                ),
            "micro_recall":
                AveragingMetric.micro_recall(
                    class_reports
                ),
        }


# 14. 실행 예시: 이진 분류
binary_labels = BinaryLabel(
    positive_label="질병 있음",
    negative_label="질병 없음"
)

actual_binary = [
    "질병 있음", "질병 있음", "질병 없음",
    "질병 없음", "질병 있음", "질병 없음",
    "질병 있음", "질병 없음"
]

probabilities = [
    0.92, 0.81, 0.72, 0.15,
    0.44, 0.20, 0.67, 0.55

전체 코드 상세 설명

1. PredictionResult

실제값, 예측값, 양성 확률을 한 객체에 저장합니다. 혼동행렬 구축의 기본 자료 단위입니다.

2. BinaryLabel

어떤 클래스를 양성으로 볼지 정의하고 원래 레이블을 양성·음성으로 변환합니다.

3. BinaryConfusionMatrix

각 예측 결과를 TP, TN, FP, FN으로 분류해 누적하고 사전 형태로 반환합니다.

4. SafeDivision

분모가 0인 경우 오류 대신 0을 반환해 지표 계산을 안전하게 처리합니다.

5. BinaryMetricCalculator

정확도, 정밀도, 재현율, 특이도, 거짓양성률, 거짓음성률, 음성 예측도, F1, 균형 정확도를 계산합니다.

6. ThresholdClassifier

양성 확률과 임계값을 비교해 양성 또는 음성 클래스를 결정합니다.

7. ThresholdAnalyzer

여러 임계값에서 혼동행렬과 모든 평가 지표를 다시 계산하여 정밀도·재현율의 변화를 비교합니다.

8. MulticlassConfusionMatrix

세 개 이상의 클래스에서 실제 클래스를 행, 예측 클래스를 열로 두어 행렬을 구성합니다.

9. OneVsRestEvaluator

관심 클래스 하나를 양성, 나머지를 음성으로 바꾸어 클래스별 TP, TN, FP, FN과 평가 지표를 계산합니다.

10. AveragingMetric

클래스별 점수의 매크로 평균, 자료 수를 반영한 가중 평균, 합산 결과 기반 마이크로 평균을 계산합니다.

11. ImbalanceAnalyzer

클래스 분포, 다수 클래스 기준 정확도, 다수·소수 클래스의 불균형 비율을 계산합니다.

12. ErrorCostPolicy

거짓양성과 거짓음성에 서로 다른 비용을 설정하여 실제 문제의 위험도를 평가에 반영합니다.

13. CostSensitiveEvaluator

FP·FN 개수와 오류 비용을 곱해 총비용과 자료당 평균 오류 비용을 계산합니다.

14. ClassificationReport

다중 클래스 혼동행렬, 전체 정확도, 클래스별 지표, 매크로·가중·마이크로 평균을 하나의 보고서로 통합합니다.

15. 이진 분류 실행

예측 확률을 임계값 0.5로 분류하고 혼동행렬·평가 지표·임계값 비교·오류 비용을 출력합니다.

16. 다중 분류 실행

A·B·C 클래스의 혼동행렬을 만들고 클래스별 One-vs-Rest 평가와 여러 평균 지표를 출력합니다.

17. 클래스 불균형 해석

음성 95개와 양성 5개 자료에서 다수 클래스만 예측해도 정확도 0.95가 됨을 확인하고 정확도의 한계를 설명합니다.

18. 객체지향 설계

자료, 행렬, 지표, 임계값, 다중분류, 평균, 비용을 독립 클래스로 나누어 재사용성과 확장성을 높였습니다.

전체 실행 흐름: 실제값과 확률 준비 → 임계값으로 예측 → 이진 혼동행렬 생성 → 지표 계산 → 임계값 비교 → 오류 비용 평가 → 다중 클래스 혼동행렬 → 클래스별 평가 → 매크로·가중·마이크로 평균 → 불균형 분석
