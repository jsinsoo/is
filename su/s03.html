4. 정답 포함 전체 코드

from collections import deque
from typing import Any


class Stack:
    """리스트를 이용한 스택 구현."""

    def __init__(self) -> None:
        self.items: list[Any] = []

    def push(self, value: Any) -> None:
        """스택의 맨 위에 자료를 삽입한다."""
        self.items.append(value)

    def pop(self) -> Any:
        """스택의 맨 위 자료를 삭제하고 반환한다."""
        if self.is_empty():
            raise IndexError("빈 스택에서는 pop할 수 없습니다.")
        return self.items.pop()

    def peek(self) -> Any:
        """스택의 맨 위 자료를 삭제하지 않고 반환한다."""
        if self.is_empty():
            raise IndexError("빈 스택에서는 peek할 수 없습니다.")
        return self.items[-1]

    def is_empty(self) -> bool:
        """스택이 비어 있는지 확인한다."""
        return len(self.items) == 0

    def size(self) -> int:
        """스택에 저장된 자료의 수를 반환한다."""
        return len(self.items)

    def snapshot(self) -> list[Any]:
        """현재 스택 상태의 복사본을 반환한다."""
        return self.items.copy()


class Queue:
    """deque를 이용한 큐 구현."""

    def __init__(self) -> None:
        self.items: deque[Any] = deque()

    def enqueue(self, value: Any) -> None:
        """큐의 뒤쪽에 자료를 삽입한다."""
        self.items.append(value)

    def dequeue(self) -> Any:
        """큐의 앞쪽 자료를 삭제하고 반환한다."""
        if self.is_empty():
            raise IndexError("빈 큐에서는 dequeue할 수 없습니다.")
        return self.items.popleft()

    def front(self) -> Any:
        """큐의 앞쪽 자료를 삭제하지 않고 반환한다."""
        if self.is_empty():
            raise IndexError("빈 큐에서는 front를 확인할 수 없습니다.")
        return self.items[0]

    def is_empty(self) -> bool:
        """큐가 비어 있는지 확인한다."""
        return len(self.items) == 0

    def size(self) -> int:
        """큐에 저장된 자료의 수를 반환한다."""
        return len(self.items)

    def snapshot(self) -> list[Any]:
        """현재 큐 상태를 리스트로 변환하여 반환한다."""
        return list(self.items)


def show_stack_state(action: str, stack: Stack) -> None:
    """스택의 현재 상태를 시각적으로 출력한다."""
    print(f"{action:<18} -> {stack.snapshot()}")

    if stack.is_empty():
        print("   [비어 있음]")
    else:
        print("   TOP")
        for value in reversed(stack.snapshot()):
            print(f"  ┌───────┐")
            print(f"  │ {str(value):^5} │")
            print(f"  └───────┘")


def show_queue_state(action: str, queue: Queue) -> None:
    """큐의 현재 상태를 시각적으로 출력한다."""
    print(f"{action:<18} -> {queue.snapshot()}")

    if queue.is_empty():
        print("   FRONT [비어 있음] REAR")
    else:
        values = " | ".join(str(value) for value in queue.snapshot())
        print(f"   FRONT ← [ {values} ] ← REAR")


stack = Stack()
queue = Queue()

print("=" * 50)
print("[스택 상태 변화]")
print("=" * 50)

for value in [10, 20, 30]:
    stack.push(value)
    show_stack_state(f"push({value})", stack)

removed_stack = stack.pop()
show_stack_state(f"pop() = {removed_stack}", stack)

top_value = stack.peek()
show_stack_state(f"peek() = {top_value}", stack)

print("\n" + "=" * 50)
print("[큐 상태 변화]")
print("=" * 50)

for value in [10, 20, 30]:
    queue.enqueue(value)
    show_queue_state(f"enqueue({value})", queue)

removed_queue = queue.dequeue()
show_queue_state(f"dequeue() = {removed_queue}", queue)

front_value = queue.front()
show_queue_state(f"front() = {front_value}", queue)

print("\n" + "=" * 50)
print("[예외 처리 확인]")
print("=" * 50)

empty_stack = Stack()
empty_queue = Queue()

for operation in [
    lambda: empty_stack.pop(),
    lambda: empty_stack.peek(),
    lambda: empty_queue.dequeue(),
    lambda: empty_queue.front(),
]:
    try:
        operation()
    except IndexError as error:
        print(error)

7. 예시 정답

문항 1 정답
스택은 마지막에 삽입한 자료가 먼저 삭제되는 LIFO 구조이고, 큐는 먼저 삽입한 자료가 먼저 삭제되는 FIFO 구조이다.
문항 2 정답
push는 스택의 맨 위에 자료를 삽입하고, pop은 맨 위 자료를 삭제하면서 반환하며, peek는 맨 위 자료를 삭제하지 않고 확인한다.
문항 3 정답
enqueue는 큐의 뒤쪽에 자료를 삽입하고, dequeue는 큐의 앞쪽 자료를 삭제하면서 반환하며, front는 앞쪽 자료를 삭제하지 않고 확인한다.
문항 4 정답
비어 있는 자료구조에는 삭제하거나 조회할 자료가 없으므로 연산을 그대로 수행하면 프로그램 오류가 발생할 수 있다. 예외 처리를 통해 잘못된 연산을 명확하게 알리고 프로그램이 예측 가능한 방식으로 동작하도록 해야 한다.
문항 5 정답
스택은 실행 취소, 괄호 검사, 함수 호출 관리 등에 활용된다. 큐는 프린터 대기열, 작업 예약, 메시지 처리, 너비 우선 탐색 등에 활용된다.
