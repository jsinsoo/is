Ⅰ. 반복구조와 재귀구조 학습자료

학습 목표
같은 팩토리얼 문제를 반복구조와 재귀구조로 각각 해결하고, 실행 결과, 실행 시간, 중앙값, 최대 메모리 사용량을 근거로 두 방식의 효율성을 평가한다.

1. 두 방식의 특징

반복구조

for문을 이용하여 2부터 n까지 차례로 곱한다.

함수 호출이 중첩되지 않으므로 호출 스택 부담이 작고 큰 입력에서 안정적이다.

재귀구조

n! = n × (n-1)!이라는 수학적 정의를 함수가 자신을 호출하는 방식으로 표현한다.

구조가 직관적이지만 입력값이 증가하면 재귀 호출 깊이와 호출 비용이 증가한다.

2. 성능 비교 기준

평가 항목 반복구조 재귀구조 평가 관점
계산 결과 정확 정확 같은 n에서 결과가 일치하는지 확인
시간 복잡도 O(n) O(n) 두 방식 모두 n에 비례하여 곱셈 수행
함수 호출 비용 낮음 매 단계 함수 호출 발생 재귀 방식의 추가 오버헤드 확인
공간 복잡도 O(1) 수준 O(n) 재귀 호출 스택 사용 여부 비교
재귀 깊이 제한 영향 없음 제한 있음 매우 큰 n에서 RecursionError 가능
표현의 직관성 절차가 명확함 수학적 정의와 유사함 문제 구조에 따른 코드 가독성 평가

3. 효율성 평가 전체 코드

평균뿐 아니라 중앙값과 최솟값도 측정하여 매우 짧은 실행 시간에서 발생할 수 있는 측정 흔들림을 함께 확인합니다.

import math
import sys
import time
import tracemalloc
from statistics import mean, median

# 재귀 호출 최대 깊이를 기본값보다 크게 설정한다.
sys.setrecursionlimit(2000)


def factorial_iter(n: int) -> int:
    """반복구조를 이용하여 n!을 계산한다."""
    if not isinstance(n, int):
        raise TypeError("n은 정수여야 합니다.")
    if n < 0:
        raise ValueError("n은 0 이상이어야 합니다.")

    result = 1

    for value in range(2, n + 1):
        result *= value

    return result


def factorial_rec(n: int) -> int:
    """재귀구조를 이용하여 n!을 계산한다."""
    if not isinstance(n, int):
        raise TypeError("n은 정수여야 합니다.")
    if n < 0:
        raise ValueError("n은 0 이상이어야 합니다.")

    if n in (0, 1):
        return 1

    return n * factorial_rec(n - 1)


def measure_time(function, n: int, repeat: int = 100) -> dict:
    """
    함수를 여러 번 실행하여 평균, 중앙값, 최솟값을 반환한다.
    매우 짧은 실행은 측정 오차가 크므로 반복 측정한다.
    """
    measured = []

    for _ in range(repeat):
        start = time.perf_counter()
        function(n)
        elapsed = time.perf_counter() - start
        measured.append(elapsed)

    return {
        "mean": mean(measured),
        "median": median(measured),
        "minimum": min(measured),
    }


def measure_peak_memory(function, n: int) -> int:
    """함수 실행 중 최대 메모리 사용량을 바이트 단위로 반환한다."""
    tracemalloc.start()

    function(n)

    current_memory, peak_memory = tracemalloc.get_traced_memory()
    tracemalloc.stop()

    return peak_memory


def compare_factorial(n: int, repeat: int = 100) -> dict:
    """반복구조와 재귀구조의 결과, 시간, 메모리를 비교한다."""
    iter_result = factorial_iter(n)
    rec_result = factorial_rec(n)

    iter_time = measure_time(factorial_iter, n, repeat)
    rec_time = measure_time(factorial_rec, n, repeat)

    iter_memory = measure_peak_memory(factorial_iter, n)
    rec_memory = measure_peak_memory(factorial_rec, n)

    return {
        "n": n,
        "same_result": iter_result == rec_result,
        "iter_mean": iter_time["mean"],
        "iter_median": iter_time["median"],
        "iter_minimum": iter_time["minimum"],
        "rec_mean": rec_time["mean"],
        "rec_median": rec_time["median"],
        "rec_minimum": rec_time["minimum"],
        "iter_memory": iter_memory,
        "rec_memory": rec_memory,
    }


inputs = [10, 100, 300, 500]
repeat = 100

print("=" * 118)
print("반복구조와 재귀구조의 팩토리얼 성능 비교")
print("=" * 118)
print(
    f"{'n':>4} | {'값 일치':>7} | "
    f"{'반복 평균(초)':>14} | {'재귀 평균(초)':>14} | "
    f"{'반복 중앙값':>14} | {'재귀 중앙값':>14} | "
    f"{'반복 메모리(KB)':>16} | {'재귀 메모리(KB)':>16}"
)
print("-" * 118)

results = []

for n in inputs:
    comparison = compare_factorial(n, repeat)
    results.append(comparison)

    print(
        f"{comparison['n']:>4} | "
        f"{str(comparison['same_result']):>7} | "
        f"{comparison['iter_mean']:>14.8f} | "
        f"{comparison['rec_mean']:>14.8f} | "
        f"{comparison['iter_median']:>14.8f} | "
        f"{comparison['rec_median']:>14.8f} | "
        f"{comparison['iter_memory'] / 1024:>16.2f} | "
        f"{comparison['rec_memory'] / 1024:>16.2f}"
    )

print("-" * 118)

# 표준 라이브러리와 결과 비교
for n in inputs:
    assert factorial_iter(n) == math.factorial(n)
    assert factorial_rec(n) == math.factorial(n)

print("모든 계산 결과가 math.factorial()과 일치합니다.")


# 오류 입력 검사
print("\n[오류 입력 검사]")

for value in [-1, 3.5, "10"]:
    for function in (factorial_iter, factorial_rec):
        try:
            function(value)
        except (TypeError, ValueError) as error:
            print(f"{function.__name__}({value!r}) -> {error}")

7. 예시 정답

문항 1 정답
두 함수 모두 0!과 1!을 1로 정의하고, 2부터 n까지의 정수를 모두 한 번씩 곱한다. 계산 절차의 표현 방식만 다르므로 최종 결과는 같다.
문항 2 정답
두 방식 모두 곱셈 횟수는 n에 비례하지만 재귀 방식은 각 단계에서 새로운 함수 호출, 매개변수 전달, 반환 주소와 지역 상태 저장이 추가로 발생한다. 따라서 같은 O(n)이라도 실제 실행 시간의 상수 비용은 재귀 방식이 더 클 수 있다.
문항 3 정답
재귀 방식은 각 호출이 끝날 때까지 이전 호출의 매개변수, 지역 변수, 반환 위치를 호출 스택에 보관한다. n이 증가하면 스택 프레임이 n단계 가까이 쌓이므로 추가 메모리가 필요하다. 반복 방식은 하나의 함수 안에서 변수 값을 갱신하므로 추가 공간이 상대적으로 작다.
문항 4 정답
매우 짧은 실행 시간은 운영체제의 작업 전환이나 다른 프로그램의 실행 상태에 영향을 받는다. 평균은 큰 이상값의 영향을 받을 수 있으므로 중앙값을 함께 보면 대표적인 실행 시간을 더 안정적으로 판단할 수 있다.
문항 5 정답
매우 큰 n에서는 반복구조가 더 적절하다. 반복구조는 재귀 깊이 제한을 받지 않고 호출 스택이 계속 쌓이지 않으며 함수 호출 오버헤드도 작다. 재귀구조는 수학적 정의를 표현하는 데 직관적이지만 큰 입력에서는 RecursionError와 메모리 증가 가능성이 있다.

9. 종합 결론

정확성: 반복구조와 재귀구조는 같은 팩토리얼 정의를 사용하므로 결과가 같다.

시간 효율성: 두 방식 모두 O(n)이지만, 재귀구조는 함수 호출 오버헤드 때문에 일반적으로 더 느리다.

공간 효율성: 반복구조는 추가 공간 사용이 작지만, 재귀구조는 호출 스택 때문에 O(n)의 추가 공간을 사용한다.

활용 판단: 수학적 정의를 이해하거나 재귀적 문제 구조를 표현할 때는 재귀구조가 유용하다. 큰 입력을 안정적으로 처리하는 실제 계산에서는 반복구조가 더 적절하다.
