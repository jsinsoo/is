import math
from collections import Counter
from dataclasses import dataclass
from typing import Any, Optional


# 1. 한 개의 학습 자료를 객체로 표현
@dataclass
class TrainingSample:
    features: list[float]
    label: Any


# 2. 전체 자료 집합 관리
class Dataset:
    def __init__(self, samples: list[TrainingSample],
                 feature_names: list[str]):
        self.samples = samples
        self.feature_names = feature_names

    @property
    def X(self) -> list[list[float]]:
        return [sample.features for sample in self.samples]

    @property
    def y(self) -> list[Any]:
        return [sample.label for sample in self.samples]

    def __len__(self) -> int:
        return len(self.samples)


# 3. 결측값 전처리
class DataPreprocessor:
    def fill_missing(self, rows: list[list[Optional[float]]],
                     replacement: float) -> list[list[float]]:
        return [
            [replacement if value is None else value for value in row]
            for row in rows
        ]


# 4. 클래스 개수와 분포 계산
class ClassCounter:
    def count(self, labels: list[Any]) -> dict[Any, int]:
        return dict(Counter(labels))

    def majority(self, labels: list[Any]) -> Any:
        return Counter(labels).most_common(1)[0][0]

    def probabilities(self, labels: list[Any]) -> dict[Any, float]:
        counts = Counter(labels)
        total = len(labels)
        return {label: count / total for label, count in counts.items()}


# 5. 지니 불순도와 엔트로피
class ImpurityCalculator:
    def gini(self, labels: list[Any]) -> float:
        probabilities = ClassCounter().probabilities(labels)
        return 1.0 - sum(p ** 2 for p in probabilities.values())

    def entropy(self, labels: list[Any]) -> float:
        probabilities = ClassCounter().probabilities(labels)
        return -sum(
            p * math.log2(p)
            for p in probabilities.values()
            if p > 0
        )


# 6. 분기 규칙
@dataclass
class SplitRule:
    feature_index: int
    threshold: float

    def go_right(self, row: list[float]) -> bool:
        return row[self.feature_index] >= self.threshold

    def describe(self, feature_names: list[str]) -> str:
        feature = feature_names[self.feature_index]
        return f"{feature} >= {self.threshold}"


# 7. 분기 후보와 정보 이득
@dataclass
class SplitCandidate:
    rule: SplitRule
    gain: float
    left_indices: list[int]
    right_indices: list[int]


class InformationGain:
    def calculate(self, parent_labels: list[Any],
                  left_labels: list[Any],
                  right_labels: list[Any],
                  criterion: str = "gini") -> float:
        calculator = ImpurityCalculator()
        impurity = (
            calculator.gini if criterion == "gini"
            else calculator.entropy
        )

        parent_impurity = impurity(parent_labels)
        total = len(parent_labels)

        left_weight = len(left_labels) / total
        right_weight = len(right_labels) / total

        children_impurity = (
            left_weight * impurity(left_labels) +
            right_weight * impurity(right_labels)
        )
        return parent_impurity - children_impurity


# 8. 최적 분기 탐색
class BestSplitFinder:
    def __init__(self, criterion: str = "gini"):
        self.criterion = criterion
        self.gain_calculator = InformationGain()

    def find(self, X: list[list[float]],
             y: list[Any]) -> Optional[SplitCandidate]:
        best: Optional[SplitCandidate] = None
        feature_count = len(X[0])

        for feature_index in range(feature_count):
            values = sorted(set(row[feature_index] for row in X))

            if len(values) < 2:
                continue

            thresholds = [
                (values[i] + values[i + 1]) / 2
                for i in range(len(values) - 1)
            ]

            for threshold in thresholds:
                rule = SplitRule(feature_index, threshold)
                left_indices = [
                    i for i, row in enumerate(X)
                    if not rule.go_right(row)
                ]
                right_indices = [
                    i for i, row in enumerate(X)
                    if rule.go_right(row)
                ]

                if not left_indices or not right_indices:
                    continue

                left_labels = [y[i] for i in left_indices]
                right_labels = [y[i] for i in right_indices]

                gain = self.gain_calculator.calculate(
                    y, left_labels, right_labels, self.criterion
                )

                candidate = SplitCandidate(
                    rule, gain, left_indices, right_indices
                )

                if best is None or candidate.gain > best.gain:
                    best = candidate

        return best


# 9. 사전 가지치기와 정지 조건
@dataclass
class TreeConfig:
    max_depth: int = 5
    min_samples_split: int = 2
    min_samples_leaf: int = 1
    min_gain: float = 0.0
    criterion: str = "gini"


class StopCondition:
    def __init__(self, config: TreeConfig):
        self.config = config

    def should_stop(self, depth: int,
                    labels: list[Any]) -> bool:
        pure = len(set(labels)) == 1
        too_deep = depth >= self.config.max_depth
        too_small = len(labels) < self.config.min_samples_split
        return pure or too_deep or too_small


# 10. 루트·내부·잎 노드
class Node:
    def __init__(self,
                 rule: Optional[SplitRule] = None,
                 prediction: Any = None,
                 probabilities: Optional[dict[Any, float]] = None,
                 depth: int = 0):
        self.rule = rule
        self.prediction = prediction
        self.probabilities = probabilities or {}
        self.depth = depth
        self.left: Optional["Node"] = None
        self.right: Optional["Node"] = None

    @property
    def is_leaf(self) -> bool:
        return self.rule is None


# 11. 객체지향 의사결정트리 분류기
class DecisionTreeClassifierOOP:
    def __init__(self, config: TreeConfig):
        self.config = config
        self.root: Optional[Node] = None
        self.feature_names: list[str] = []
        self.feature_importance_raw: dict[int, float] = {}

    def fit(self, dataset: Dataset) -> None:
        self.feature_names = dataset.feature_names
        self.feature_importance_raw = {
            i: 0.0 for i in range(len(self.feature_names))
        }
        self.root = self._build_tree(
            dataset.X, dataset.y, depth=0
        )

    def _build_tree(self, X: list[list[float]],
                    y: list[Any],
                    depth: int) -> Node:
        counter = ClassCounter()
        node = Node(
            prediction=counter.majority(y),
            probabilities=counter.probabilities(y),
            depth=depth
        )

        stop = StopCondition(self.config)
        if stop.should_stop(depth, y):
            return node

        finder = BestSplitFinder(self.config.criterion)
        best = finder.find(X, y)

        if best is None or best.gain <= self.config.min_gain:
            return node

        if (len(best.left_indices) < self.config.min_samples_leaf or
                len(best.right_indices) < self.config.min_samples_leaf):
            return node

        node.rule = best.rule
        self.feature_importance_raw[best.rule.feature_index] += (
            best.gain * len(y)
        )

        left_X = [X[i] for i in best.left_indices]
        left_y = [y[i] for i in best.left_indices]
        right_X = [X[i] for i in best.right_indices]
        right_y = [y[i] for i in best.right_indices]

        node.left = self._build_tree(left_X, left_y, depth + 1)
        node.right = self._build_tree(right_X, right_y, depth + 1)

        return node

    def predict_one(self, row: list[float]) -> Any:
        if self.root is None:
            raise RuntimeError("먼저 fit()으로 모델을 학습해야 합니다.")

        node = self.root
        while not node.is_leaf:
            if node.rule.go_right(row):
                node = node.right
            else:
                node = node.left
        return node.prediction

    def predict(self, X: list[list[float]]) -> list[Any]:
        return [self.predict_one(row) for row in X]

    def predict_proba_one(self, row: list[float]) -> dict[Any, float]:
        if self.root is None:
            raise RuntimeError("먼저 fit()으로 모델을 학습해야 합니다.")

        node = self.root
        while not node.is_leaf:
            node = node.right if node.rule.go_right(row) else node.left
        return node.probabilities

    def feature_importances(self) -> dict[str, float]:
        total = sum(self.feature_importance_raw.values())

        if total == 0:
            return {name: 0.0 for name in self.feature_names}

        return {
            self.feature_names[index]:
            self.feature_importance_raw[index] / total
            for index in range(len(self.feature_names))
        }

    def print_tree(self) -> None:
        if self.root is None:
            print("학습된 트리가 없습니다.")
            return
        self._print_node(self.root)

    def _print_node(self, node: Node,
                    indent: str = "") -> None:
        if node.is_leaf:
            print(
                f"{indent}잎: 예측={node.prediction}, "
                f"확률={node.probabilities}"
            )
            return

        question = node.rule.describe(self.feature_names)
        print(f"{indent}[질문] {question}")
        print(f"{indent}├─ 아니오")
        self._print_node(node.left, indent + "│  ")
        print(f"{indent}└─ 예")
        self._print_node(node.right, indent + "   ")

# 13. 정확도와 혼동행렬
class ClassificationEvaluator:
    def accuracy(self, y_true: list[Any],
                 y_pred: list[Any]) -> float:
        correct = sum(
            actual == predicted
            for actual, predicted in zip(y_true, y_pred)
        )
        return correct / len(y_true)

    def confusion_matrix(self, y_true: list[Any],
                         y_pred: list[Any]) -> dict:
        labels = sorted(set(y_true) | set(y_pred))
        matrix = {
            actual: {predicted: 0 for predicted in labels}
            for actual in labels
        }

        for actual, predicted in zip(y_true, y_pred):
            matrix[actual][predicted] += 1

        return matrix


# 14. 사후 가지치기 판단
class PruningDecision:
    def should_prune(self,
                     validation_before: float,
                     validation_after: float) -> bool:
        return validation_after >= validation_before


# 15. 분류 트리와 회귀 트리의 공통 인터페이스
class BaseTreeModel:
    def fit(self, dataset: Dataset) -> None:
        raise NotImplementedError

    def predict(self, X: list[list[float]]) -> list[Any]:
        raise NotImplementedError


class ClassificationTreeModel(BaseTreeModel):
    def __init__(self, config: TreeConfig):
        self.model = DecisionTreeClassifierOOP(config)

    def fit(self, dataset: Dataset) -> None:
        self.model.fit(dataset)

    def predict(self, X: list[list[float]]) -> list[Any]:
        return self.model.predict(X)


class RegressionTreeModel(BaseTreeModel):
    def fit(self, dataset: Dataset) -> None:
        self.mean_value = sum(dataset.y) / len(dataset.y)

    def predict(self, X: list[list[float]]) -> list[float]:
        return [self.mean_value for _ in X]


# 16. 전체 학습 시스템: 합성
class LearningSystem:
    def __init__(self,
                 model: BaseTreeModel,
                 evaluator: ClassificationEvaluator):
        self.model = model
        self.evaluator = evaluator

    def train(self, dataset: Dataset) -> None:
        self.model.fit(dataset)

    def evaluate(self, dataset: Dataset) -> float:
        predictions = self.model.predict(dataset.X)
        return self.evaluator.accuracy(
            dataset.y, predictions
        )

전체 코드 상세 설명

1. 자료 객체

TrainingSample은 한 행의 특성과 레이블을 저장하고, Dataset은 여러 객체를 모아 입력 자료와 정답을 제공한다.

2. 전처리

DataPreprocessor는 결측값을 대체한다. 실제 프로젝트에서는 평균·중앙값·최빈값 등 자료 특성에 맞는 방법을 선택해야 한다.

3. 클래스 분포

ClassCounter는 클래스별 개수, 다수 클래스, 클래스 확률을 계산한다. 잎 노드의 예측값과 확률을 정하는 데 사용된다.

4. 불순도

ImpurityCalculator는 지니 불순도와 엔트로피를 모두 지원한다. 설정에 따라 분기 품질 평가 기준을 바꿀 수 있다.

5. 분기 규칙

SplitRule은 특성 인덱스와 임계값을 캡슐화한다. 입력값이 임계값 이상이면 오른쪽, 미만이면 왼쪽으로 이동한다.

6. 정보 이득

InformationGain은 부모 불순도에서 자식 노드의 가중 불순도를 뺀다. 값이 클수록 분기 후 클래스가 더 잘 구분된다.

7. 최적 분기

BestSplitFinder는 모든 특성의 가능한 임계값을 시험하고 정보 이득이 가장 큰 후보를 선택한다.

8. 정지 조건

TreeConfig와 StopCondition은 최대 깊이, 최소 분할 자료 수, 최소 잎 자료 수, 최소 정보 이득을 관리한다.

9. 노드 객체

Node는 내부 노드와 잎 노드를 하나의 클래스로 표현한다. rule이 없으면 잎 노드이다.

10. 재귀 학습

_build_tree()는 정지 조건을 검사하고 최적 분기를 찾은 뒤 왼쪽과 오른쪽 자식 노드를 재귀적으로 생성한다.

11. 예측

predict_one()은 루트에서 시작해 분기 규칙을 반복 적용하여 잎 노드에 도달한 뒤 다수 클래스를 반환한다.

12. 확률 예측

predict_proba_one()은 잎 노드에 포함된 클래스의 비율을 반환하여 예측의 확신 정도를 보여 준다.

13. 특성 중요도

각 분기의 정보 이득과 자료 수를 누적해 정규화한다. 값이 크면 모델이 그 특성을 많이 사용했다는 뜻이지만 인과관계는 아니다.

14. 트리 시각화

print_tree()는 질문, 왼쪽 가지, 오른쪽 가지, 잎의 예측값과 확률을 텍스트 구조로 출력한다.

15. 평가

ClassificationEvaluator는 정확도와 혼동행렬을 계산한다. 정확도는 전체 성능을, 혼동행렬은 오류 유형을 보여 준다.

16. 가지치기

PruningDecision은 가지 제거 전후의 검증 성능을 비교한다. 사전 가지치기는 학습 중, 사후 가지치기는 학습 후 수행한다.

17. 상속과 다형성

BaseTreeModel을 상속한 분류·회귀 모델은 같은 fit(), predict() 인터페이스를 서로 다르게 구현한다.

18. 합성

LearningSystem은 모델과 평가기 객체를 내부에 포함하여 학습과 평가 기능을 연결한다.

실행 결과 해석: 예제 자료로 모델을 학습한 뒤 학습 정확도, 텍스트 트리, 새 학생 예측, 클래스 확률, 특성 중요도, 혼동행렬, 사후 가지치기 판단을 차례로 출력합니다.
