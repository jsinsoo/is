배점: 옳은 것 3개 고르기 10문항×4점=40점, OX 10문항×3점=30점, 옳은 것 1개 고르기 10문항×3점=30점. 다중선택은 정답 3개를 모두 맞혀야 득점합니다.

코드 A — 반복문을 이용한 3:4:5 피타고라스 나무

import turtle

t = turtle.Turtle()
t.speed(0)
t.hideturtle()

def square(A, B):
    x1, y1 = A
    x2, y2 = B
    dx = x2 - x1
    dy = y2 - y1

    C = (x2 - dy, y2 + dx)
    D = (x1 - dy, y1 + dx)

    t.penup()
    t.goto(A)
    t.pendown()
    t.goto(B)
    t.goto(C)
    t.goto(D)
    t.goto(A)

    return D, C

squares = [((-70, -330), (70, -330))]

for _ in range(9):
    next_squares = []

    for A, B in squares:
        L, R = square(A, B)

        dx = R[0] - L[0]
        dy = R[1] - L[1]

        P = (
            L[0] + 16/25 * dx - 12/25 * dy,
            L[1] + 16/25 * dy + 12/25 * dx
        )

        next_squares.append((L, P))
        next_squares.append((P, R))

    squares = next_squares

turtle.done()

코드 B — 재귀함수를 이용한 3:4:5 피타고라스 나무

import turtle

t = turtle.Turtle()
t.speed(0)
t.hideturtle()

screen = turtle.Screen()
screen.setup(900, 800)

def square(A, B):
    x1, y1 = A
    x2, y2 = B
    dx = x2 - x1
    dy = y2 - y1

    C = (x2 - dy, y2 + dx)
    D = (x1 - dy, y1 + dx)

    t.penup()
    t.goto(A)
    t.pendown()
    t.goto(B)
    t.goto(C)
    t.goto(D)
    t.goto(A)

    return D, C

def tree(A, B, depth):
    if depth == 0:
        return

    L, R = square(A, B)

    dx = R[0] - L[0]
    dy = R[1] - L[1]

    P = (
        L[0] + 16/25 * dx - 12/25 * dy,
        L[1] + 16/25 * dy + 12/25 * dx
    )

    tree(L, P, depth - 1)
    tree(P, R, depth - 1)

tree((-70, -330), (70, -330), 9)

turtle.done()
