def catalan(n):
    if n in memo:
        return memo[n]

    return total

1점

memo = {}

def matrix_chain(i, j):
    if i == j:
        return 0
    key = (i, j)
    if key in memo:
        return memo[key]
    best = float("inf")
    for k in range(i, j):
        cost = (matrix_chain(i, k)
                + matrix_chain(k + 1, j)
                + dims[i - 1] * dims[k] * dims[j])
        best = min(best, cost)
    memo[key] = best
    return best

print("최소 스칼라 곱셈 횟수:", matrix_chain(1, len(dims)-1))

    if k < 0 or k > n:
        return 0.0
    if n == 0:
        return 1.0 if k == 0 else 0.0

1점

        return 0.0

        return memo[i]
    r = (indices[i] - indices[i+1]) / (indices[i] + indices[i+1])
