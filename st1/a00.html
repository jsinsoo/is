from collections import deque def bfs(graph, start): visited = [False] * len(graph) q = deque([start]) visited[start] = True order = [] while q: v = q.popleft() order.append(v) for nxt in graph[v]: if not visited[nxt]: visited[nxt] = True q.append(nxt) return order def dfs(graph, v, visited, order): visited[v] = True order.append(v) for nxt in graph[v]: if not visited[nxt]: dfs(graph, nxt, visited, order)
